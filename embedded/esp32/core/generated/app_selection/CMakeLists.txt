file(GLOB APP_COMPONENT_DIRS LIST_DIRECTORIES true
    "${CMAKE_SOURCE_DIR}/apps/*"
)

set(APP_COMPONENT_REQUIRES "")
set(APP_COMPONENT_INCLUDE_DIRS "")
foreach(APP_COMPONENT_DIR ${APP_COMPONENT_DIRS})
    if(IS_DIRECTORY "${APP_COMPONENT_DIR}" AND EXISTS "${APP_COMPONENT_DIR}/CMakeLists.txt")
        get_filename_component(APP_COMPONENT_NAME "${APP_COMPONENT_DIR}" NAME)
        list(APPEND APP_COMPONENT_REQUIRES "${APP_COMPONENT_NAME}")
        list(APPEND APP_COMPONENT_INCLUDE_DIRS "${APP_COMPONENT_DIR}")
    endif()
endforeach()

idf_component_register(
    SRCS "app_selection.c"
    INCLUDE_DIRS "." ${APP_COMPONENT_INCLUDE_DIRS}
    REQUIRES ${APP_COMPONENT_REQUIRES}
    WHOLE_ARCHIVE
)
